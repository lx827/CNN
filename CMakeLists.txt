cmake_minimum_required(VERSION 3.28)
if(CMAKE_HOST_WIN32 AND NOT DEFINED CMAKE_CXX_COMPILER)
    set(CMAKE_CXX_COMPILER cl)
endif()
if(CMAKE_HOST_WIN32 AND NOT DEFINED CMAKE_C_COMPILER)
    set(CMAKE_C_COMPILER cl)
endif()
project(cnn_compute VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
if(SKBUILD)
    set(CNN_PYTHON_DEFAULT ON)
else()
    set(CNN_PYTHON_DEFAULT OFF)
endif()
option(CNN_COMPUTE_BUILD_PYTHON "Build Python bindings" ${CNN_PYTHON_DEFAULT})
if(CNN_COMPUTE_BUILD_PYTHON)
    set(CNN_SHARED_DEFAULT OFF)
else()
    set(CNN_SHARED_DEFAULT ON)
endif()
option(CNN_COMPUTE_BUILD_SHARED "Build the C ABI shared library" ${CNN_SHARED_DEFAULT})
if(WIN32 AND NOT MSVC)
    message(FATAL_ERROR "Windows builds require MSVC; run uv sync through vsdev")
endif()
if(MSVC)
    include(cmake/DetectMsvcIncludes.cmake)
    cnn_compute_detect_msvc_includes()
endif()

add_subdirectory(lib)
if(CNN_COMPUTE_BUILD_SHARED)
    add_subdirectory(capi)
endif()
if(NOT SKBUILD)
    include(cmake/install_native.cmake)
endif()
if(CNN_COMPUTE_BUILD_PYTHON)
    add_subdirectory(bind)
endif()

option(CNN_COMPUTE_BUILD_TESTS "Build native C++ interface contract tests" OFF)
if(CNN_COMPUTE_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests/native)
    if(CNN_COMPUTE_BUILD_SHARED)
        add_subdirectory(capi/tests)
    endif()
endif()
