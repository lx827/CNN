cmake_minimum_required(VERSION 3.28)
project(cnn_native_consumer LANGUAGES C CXX)
find_package(CnnCompute 0.1.0 EXACT CONFIG REQUIRED)
add_executable(c_consumer ../../capi/tests/consumer.c)
target_link_libraries(c_consumer PRIVATE CNN::c)
target_compile_features(c_consumer PRIVATE c_std_11)
if(UNIX)
    target_link_libraries(c_consumer PRIVATE m)
endif()
add_executable(cpp_consumer main.cpp)
target_link_libraries(cpp_consumer PRIVATE CNN::wavelet CNN::numeric CNN::sequence)
enable_testing()
add_test(NAME installed_c_consumer COMMAND c_consumer)
add_test(NAME installed_cpp_consumer COMMAND cpp_consumer)
