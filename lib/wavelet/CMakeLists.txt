enable_language(C)
add_library(cnn_compute_pywt STATIC vendor/pywt/common.c vendor/pywt/convolution.c
            vendor/pywt/wavelets.c vendor/pywt/wt.c)
set_target_properties(cnn_compute_pywt PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    target_compile_options(cnn_compute_pywt PRIVATE /utf-8 /fp:strict)
else()
    target_compile_options(cnn_compute_pywt PRIVATE -ffp-contract=off)
endif()

target_link_libraries(cnn_compute_wavelet PUBLIC CNN::core PRIVATE cnn_compute_pywt)
